# Packed xcrsarfrun config.
# This avoids extra XSPEC source slots for the observed region sources. XSPEC
# evaluates one xpack model over all data groups, and xcrsarf internally mixes
# source-region spectra into each target spectrum using the ARF map below.
# A source with a model block but no spectrum line is added afterward as a
# normal source-only XSPEC model, which is useful for point sources, AGN, etc.

mode packed
sources 3

# Source 1 is the packed default. Sources without an explicit model block
# inherit source 1's expression and parameter rows.

model 1 apec*const
10.1 0.1 1.0 1.0 25.0 25.0
0.4 -0.01 0.1 0.1 1.0 1.0
0.0546 -0.01 0.01 0.01 0.30 0.30
0.002 1e-6 1e-6 1e-6 1e-1 1e-1
1.0 -1 0.0 0.0 2.0 2.0

model 2 apec*const
8.0 0.1 1.0 1.0 25.0 25.0
0.3 -0.01 0.1 0.1 1.0 1.0
0.0546 -0.01 0.01 0.01 0.30 0.30
0.001 1e-6 1e-6 1e-6 1e-1 1e-1
1.0 -1 0.0 0.0 2.0 2.0

# Source 3 is source-only: it has no spectrum row. In packed mode xcrsarfrun
# loads its response/ARF slots after xpack and creates model 3:s3 po*const.
model 3 po*const
1.7 0.01 0.1 0.1 3.0 3.0
1e-4 1e-7 1e-8 1e-8 1e-1 1e-1
1.0 -1 0.0 0.0 2.0 2.0

# The constant that is frozen between observations (FPMA).
reference_obs A

# spectrum <specNum> <region> <constantLabel> <pha> <rmf> <diagonalArf> [mixGroup]
# Obs 1 A
spectrum 1 1 A obsA/reg01.pha obsA/reg01.rmf obsA/reg01_1.arf
spectrum 2 1 B obsB/reg01.pha obsB/reg01.rmf obsB/reg01_1.arf
# Obs 1 B
spectrum 3 2 A obsA/reg02.pha obsA/reg02.rmf obsA/reg02_2.arf
spectrum 4 2 B obsB/reg02.pha obsB/reg02.rmf obsB/reg02_2.arf

# If A/B repeat for another observation, keep the same constant labels and add
# a unique mixGroup so cross-ARF mixing stays within that observation:
# # Obs 2 A
# spectrum 5 1 A obs002/reg01A.pha obs002/reg01A.rmf obs002/reg01A_1.arf obs002_A
# # Obs 2 B
# spectrum 6 1 B obs002/reg01B.pha obs002/reg01B.rmf obs002/reg01B_1.arf obs002_B

# arf <sourceRegion> <targetSpecNum> <arfPath>
# Obs 1 A ->Src 1
arf 1 1 obsA/reg01_1.arf
arf 2 1 obsA/reg02_1.arf
arf 3 1 obsA/ptsrc_1.arf
# Obs 1 B ->Src 1
arf 1 2 obsB/reg01_1.arf
arf 2 2 obsB/reg02_1.arf
arf 3 2 obsB/ptsrc_1.arf
# Obs 1 A ->Src 2
arf 1 3 obsA/reg01_2.arf
arf 2 3 obsA/reg02_2.arf
arf 3 3 obsA/ptsrc_2.arf
# Obs 1 B ->Src 2
arf 1 4 obsB/reg01_2.arf
arf 2 4 obsB/reg02_2.arf
arf 3 4 obsB/ptsrc_2.arf
